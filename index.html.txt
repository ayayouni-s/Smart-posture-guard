<script>
    // --- State Variables ---
    let bluetoothDevice;
    let uartCharacteristic;
    let baselinePitch = 0;
    let currentPitch = 0;
    let slouchCount = 0;
    let slouchStartTime = null;
    let countedCurrentSlouch = false;

    const SLOUCH_THRESHOLD = 15; // 15 degrees threshold
    const SLOUCH_TIME_LIMIT_MS = 5000; // 5 seconds continuous slouch triggers count increment

    // --- DOM Elements ---
    const connectBtn = document.getElementById('connectBtn');
    const calibrateBtn = document.getElementById('calibrateBtn');
    const resetCounterBtn = document.getElementById('resetCounterBtn');
    const connectionStatus = document.getElementById('connectionStatus');
    const tiltAngleDisplay = document.getElementById('tiltAngle');
    const postureStateDisplay = document.getElementById('postureState');
    const slouchCountDisplay = document.getElementById('slouchCount');

    // Reset Counter Event
    resetCounterBtn.addEventListener('click', () => {
        slouchCount = 0;
        slouchCountDisplay.textContent = slouchCount;
    });

    // --- Web Bluetooth Micro:bit Connection ---
    const UART_SERVICE_UUID = '6e400001-b5a3-f393-e0a9-e50e24dcca9e';
    const UART_TX_CHARACTERISTIC_UUID = '6e400003-b5a3-f393-e0a9-e50e24dcca9e';

    connectBtn.addEventListener('click', async () => {
        if (!navigator.bluetooth) {
            alert("Web Bluetooth is not supported on this browser/device. Try using Chrome or Edge.");
            return;
        }

        try {
            connectionStatus.textContent = "Connecting...";
            bluetoothDevice = await navigator.bluetooth.requestDevice({
                filters: [{ namePrefix: 'BBC micro:bit' }],
                optionalServices: [UART_SERVICE_UUID]
            });

            const server = await bluetoothDevice.gatt.connect();
            const service = await server.getPrimaryService(UART_SERVICE_UUID);
            uartCharacteristic = await service.getCharacteristic(UART_TX_CHARACTERISTIC_UUID);

            await uartCharacteristic.startNotifications();
            uartCharacteristic.addEventListener('characteristicvaluechanged', handleDataReceived);

            connectionStatus.textContent = "Connected";
            connectionStatus.className = "status-badge status-connected";
            connectBtn.disabled = true;
            calibrateBtn.disabled = false;

            bluetoothDevice.addEventListener('gattserverdisconnected', onDisconnected);
        } catch (error) {
            console.error("Bluetooth Connection Error:", error);
            connectionStatus.textContent = "Disconnected";
            connectionStatus.className = "status-badge status-disconnected";
        }
    });

    function onDisconnected() {
        connectionStatus.textContent = "Disconnected";
        connectionStatus.className = "status-badge status-disconnected";
        connectBtn.disabled = false;
        calibrateBtn.disabled = true;
    }

    // Zero-calibration button
    calibrateBtn.addEventListener('click', () => {
        baselinePitch = currentPitch;
        alert(`Calibrated! Current position set as zero reference.`);
    });

    // Handle incoming telemetry stream from micro:bit
    function handleDataReceived(event) {
        const decoder = new TextDecoder('utf-8');
        const rawString = decoder.decode(event.target.value).trim();
        const rawPitch = parseFloat(rawString);

        if (!isNaN(rawPitch)) {
            // Calculate pitch relative to calibrated zero angle
            currentPitch = rawPitch - baselinePitch;
            updateDashboard(currentPitch);
        }
    }

    function updateDashboard(pitch) {
        const roundedPitch = Math.round(pitch);
        tiltAngleDisplay.textContent = `${roundedPitch}°`;

        // Check posture threshold
        if (pitch > SLOUCH_THRESHOLD) {
            postureStateDisplay.textContent = "Slouching Detected!";
            postureStateDisplay.className = "posture-status bad-posture";

            if (!slouchStartTime) {
                slouchStartTime = Date.now();
            } else if (Date.now() - slouchStartTime >= SLOUCH_TIME_LIMIT_MS && !countedCurrentSlouch) {
                slouchCount++;
                slouchCountDisplay.textContent = slouchCount;
                countedCurrentSlouch = true; // Count once per slouch session
            }
        } else {
            postureStateDisplay.textContent = "Good Posture";
            postureStateDisplay.className = "posture-status good-posture";
            
            // Reset slouch tracker when posture returns to normal
            slouchStartTime = null;
            countedCurrentSlouch = false;
        }
    }
</script>
